-- Today's Scheduled Callbacks

WITH latest AS (
  SELECT DISTINCT ON (sr.student_id)
    sr.student_id,
    sr.callback_date
  FROM student_remarks sr
  WHERE sr.isdisabled = false
  ORDER BY sr.student_id, sr.created_at DESC
)
SELECT
  tow.counsellor_name AS to_name,
  l2.counsellor_name  AS l2_name,
  l2.counsellor_id,
  COUNT(*) AS scheduled_today
FROM latest lt
INNER JOIN students s        ON lt.student_id = s.student_id
INNER JOIN counsellors l2    ON s.assigned_counsellor_id = l2.counsellor_id
INNER JOIN counsellors tow   ON l2.assigned_to = tow.counsellor_id
WHERE lt.callback_date = (NOW() AT TIME ZONE 'Asia/Kolkata')::date
  AND l2.counsellor_id IN ('CNS-95E89FD2','CNS-A7C45EF3','CNS-50FC5C75','CNS-C955E4FD',
    'CNS-46211CD0','CNS-3202BBFB','CNS-F73AC472','CNS-98EBC81C','CNS-DF1C4738','CNS-F77A0C4F',
    'CNS-31F88819','CNS-41CB98F3','CNS-832F23A3','CNS-55D3DFD1','CNS-A4FEB626','CNS-044DCBE8',
    'CNS-54EF30DC','CNS-FE30B430','CNS-9B43C0C5','CNS-4A8DF2BC','CNS-E0E66B47','CNS-B67EBAD9',
    'CNS-EA7E1516','CNS-19EC286D','CNS-B20D26C2','CNS-DEC844DA','CNS-CDAD6465','CNS-24936D78',
    'CNS-3FFA343D','CNS-519AA26B','CNS-B5CA4D71','CNS-A6E3F09E','CNS-00AB3C7B','CNS-DE855747',
    'CNS-869C2811')
GROUP BY tow.counsellor_name, l2.counsellor_name, l2.counsellor_id
ORDER BY tow.counsellor_name, l2.counsellor_name
