Here are two ready-to-run queries — just paste and execute, no parameters needed.

**1. For `regular_lms`** (covers 919484958351 – CU LPU Combined, and 919484958355 – DSA):

```sql
WITH campaign_leads AS (
  SELECT DISTINCT ON (sla.student_id)
    sla.student_id, sla.utm_campaign
  FROM student_lead_activities sla
  WHERE sla.utm_campaign IN ('919484958351','919484958355')
  ORDER BY sla.student_id, sla.created_at ASC
),
first_form AS (
  SELECT cl.utm_campaign, csj.student_id, csj.course_id, MIN(csj.created_at) AS form_date
  FROM course_status_journeys csj
  JOIN campaign_leads cl ON cl.student_id = csj.student_id
  WHERE csj.course_status IN (
    'Form Submitted – Portal Pending','Form Submitted – Completed',
    'Walkin Completed','Walkin Marked','Exam/Interview Scheduled',
    'Offer Letter/Results Pending','Offer Letter/Results Released','Ready For Admission'
  )
  GROUP BY cl.utm_campaign, csj.student_id, csj.course_id
),
first_admission AS (
  SELECT cl.utm_campaign, csj.student_id, csj.course_id, MIN(csj.created_at) AS adm_date
  FROM course_status_journeys csj
  JOIN campaign_leads cl ON cl.student_id = csj.student_id
  WHERE csj.course_status IN ('Admission','Enrolled')
    AND csj.fee_type NOT IN ('Partial Done','Partial Paid','Partially Paid')
  GROUP BY cl.utm_campaign, csj.student_id, csj.course_id
)
SELECT
  cl.utm_campaign AS campaign_number,
  COUNT(DISTINCT cl.student_id) AS leads,
  COUNT(DISTINCT ff.student_id) AS applications,
  COUNT(DISTINCT fa.student_id) AS admissions
FROM campaign_leads cl
LEFT JOIN first_form ff ON ff.student_id = cl.student_id AND ff.utm_campaign = cl.utm_campaign
LEFT JOIN first_admission fa ON fa.student_id = cl.student_id AND fa.utm_campaign = cl.utm_campaign
GROUP BY cl.utm_campaign
ORDER BY cl.utm_campaign
```
